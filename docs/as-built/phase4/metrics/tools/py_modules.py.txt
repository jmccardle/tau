"""Per-module LOC / function count / CC for python/tau_next (radon). Run from the repo root with radon on PATH; writes phase4/metrics/data/python-modules.json and prints a table."""
import json,subprocess,sys,collections
from pathlib import Path
d=json.load(open("docs/as-built/phase4/metrics/data/python.json"))
p=subprocess.run(["radon","cc","-j","-s","python/tau_next"],capture_output=True,text=True,check=True)
fn=collections.defaultdict(list)
for path,bl in json.loads(p.stdout).items():
    for b in bl:
        if b["type"]!="class": fn[path].append(b["complexity"])
rows=[]
for f in sorted(Path("python/tau_next").rglob("*.py")):
    s=str(f);t=f.read_text().splitlines();c=fn.get(s,[])
    code=sum(1 for l in t if l.strip() and not l.strip().startswith("#"))
    rows.append(dict(module=s.removeprefix("python/"),lines=len(t),code_lines=code,functions=len(c),cc_max=max(c,default=0),cc_mean=round(sum(c)/len(c),2) if c else 0,over_10=sum(x>10 for x in c)))
json.dump(rows,open("docs/as-built/phase4/metrics/data/python-modules.json","w"),indent=1)
print("| module | lines | functions | CC max | CC mean | >10 |\n|---|---:|---:|---:|---:|---:|")
for r in sorted(rows,key=lambda r:-r["lines"]):print(f"| `{r['module']}` | {r['lines']} | {r['functions']} | {r['cc_max']} | {r['cc_mean']} | {r['over_10']} |")
